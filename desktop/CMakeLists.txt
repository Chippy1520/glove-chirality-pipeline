cmake_minimum_required(VERSION 3.24)
project(GRIPDesktop VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Network Quick QuickControls2 Concurrent Multimedia)
qt_standard_project_setup(REQUIRES 6.8)
qt_add_executable(GRIP WIN32 main.cpp GripClient.cpp GripClient.h)
qt_add_qml_module(GRIP URI GRIP VERSION 1.0 QML_FILES Main.qml WorkflowForm.qml Inspect.qml Tools.qml)
target_link_libraries(GRIP PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network Qt6::Quick Qt6::QuickControls2 Qt6::Concurrent Qt6::Multimedia)
install(TARGETS GRIP RUNTIME DESTINATION .)
